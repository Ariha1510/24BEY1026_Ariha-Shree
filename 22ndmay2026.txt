use vitcampus;
create table departments(
did int primary key,
dname varchar(25) not null unique
);
insert into departments values
(101,'CSE'),(102,'ECE'),(103,'civil'),(106,'Aero'),(107,'Mechanical');

select * from departments order by did;

create table HOD
(Hid int primary key, hname varchar(20) not null unique);

insert into HOD values
(101,'Arun'),(102,'Ariha'),(104,'Shree'),(105,'Anvi');

select * from HOD;

select * from departments CROSS JOIN hod order by did asc;

select * from departments INNER JOIN hod where departments.did = hod.hid;

Select * from departments NATURAL JOIN hod;

select * from departments LEFT OUTER JOIN
hod ON (departments.did = hod.hid) order by did asc;

select * from departments FULL JOIN
hod ON (departments.did = hod.hid) order by did asc;
